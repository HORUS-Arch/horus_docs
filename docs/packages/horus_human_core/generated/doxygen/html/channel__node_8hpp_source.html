323 void onSample(const std::string& ch, typename MsgT::UniquePtr msg)
324 {
325 const int mode = mode_;
326 std::unique_ptr<MsgT> raw;
327 if (mode == BYPASS) raw = std::make_unique<MsgT>(*msg); /* published untouched */
328 condition(ch, *msg); /* always: keeps stateful filters continuous across mode changes */
329 {
330 std::lock_guard<std::mutex> lock(mutex_);
331 if (collecting_.count(ch))
332 {
333 plugin_->add(ch, *msg);
334 ++collected_[ch];
335 }
336 if (plugin_ && mode == CALIBRATED) plugin_->apply(ch, *msg);
337 }
338 publish(ch, raw ? *raw : *msg);
339 }

361 void runCalibration(const std::shared_ptr<GoalHandle> handle)
362 {
363 const auto goal = handle->get_goal();
364 auto result = std::make_shared<typename Calibrate::Result>();
365 const std::vector<std::string> units =
366 goal->unit.empty() ? channels_ : std::vector<std::string>{goal->unit};
367 double duration = goal->duration;
368 {
369 std::lock_guard<std::mutex> lock(mutex_);
370 if (duration <= 0.0) duration = plugin_->defaultDuration();
371 for (const auto& u : units)
372 {
373 plugin_->begin(u);
374 collecting_.insert(u);
375 collected_[u] = 0;
376 }
377 }
378
379 auto feedback = std::make_shared<typename Calibrate::Feedback>();
380 feedback->unit = goal->unit;
381 const auto start = std::chrono::steady_clock::now();
382 bool canceled = false;
383 while (rclcpp::ok() && !stop_calibration_)
384 {
385 const double elapsed =
386 std::chrono::duration<double>(std::chrono::steady_clock::now() - start).count();
387 if (elapsed >= duration) break;
388 if (handle->is_canceling())
389 {
390 canceled = true;
391 break;
392 }
393 feedback->progress = static_cast<float>(elapsed / duration);
394 feedback->message = "collecting";
395 handle->publish_feedback(feedback);
396 std::this_thread::sleep_for(std::chrono::milliseconds(100));
397 }
398
399 std::string summary;
400 bool ok = !canceled && !stop_calibration_;
401 {
402 std::lock_guard<std::mutex> lock(mutex_);
403 for (const auto& u : units) collecting_.erase(u);
404 for (const auto& u : units)
405 {
406 if (!ok) break;
407 if (collected_[u] == 0)
408 {
409 result->error_code = Calibrate::Result::NO_DATA;
410 summary = "no samples on '" + u + "'";
411 ok = false;
412 break;
413 }
414 YAML::Node data;
415 std::string unit_summary;
416 if (!plugin_->finish(u, data, unit_summary) || !plugin_->load(u, data))
417 {
418 result->error_code = Calibrate::Result::CALIBRATION_FAILED;
419 summary = u + ": " + unit_summary;
420 ok = false;
421 break;
422 }
423 pending_[u] = data;
424 sources_[u] = "unsaved";
425 result->units.push_back(u);
426 summary += (summary.empty() ? "" : "; ") + u + ": " + unit_summary;
427 }
428 }
429
430 calibrating_ = false;
431 if (canceled)
432 {
433 handle->canceled(result);
434 return;
435 }
436 if (!ok)
437 {
438 if (result->error_code == Calibrate::Result::NONE) result->error_code = Calibrate::Result::UNKNOWN;
439 result->error_msg = summary.empty() ? "calibration interrupted" : summary;
440 if (rclcpp::ok()) handle->abort(result);
441 return;
442 }
443 result->summary = summary;
444 handle->succeed(result);
445 RCLCPP_INFO(get_logger(), "calibrated (pending save): %s", summary.c_str());
446 }

450 void onSave(const std::shared_ptr<horus_msgs::srv::SaveCalibration::Request> req,
451 std::shared_ptr<horus_msgs::srv::SaveCalibration::Response> res)
452 {
453 std::lock_guard<std::mutex> lock(mutex_);
454 const std::string subject = req->subject_id.empty() ? get_parameter("subject_id").as_string() : req->subject_id;
455 if (!store_ || subject.empty())
456 {
457 res->success = false;
458 res->message = store_ ? "no subject_id" : "node not configured";
459 return;
460 }
461 std::vector<std::string> units;
462 for (const auto& [u, data] : pending_)
463 if (req->unit.empty() || req->unit == u) units.push_back(u);
464 if (units.empty())
465 {
466 res->success = false;
467 res->message = "nothing pending to save";
468 return;
469 }
470 try
471 {
472 for (const auto& u : units)
473 {
474 res->paths.push_back(store_->save(subject, get_name(), u, plugin_name_, pending_[u]));
475 sources_[u] = res->paths.back();
476 pending_.erase(u);
477 }
478 res->success = true;
479 res->message = "saved " + std::to_string(units.size()) + " file(s)";
480 }
481 catch (const std::exception& e)
482 {
483 res->success = false;
484 res->message = e.what();
485 }
486 }

488 void onLoad(const std::shared_ptr<horus_msgs::srv::LoadCalibration::Request> req,
489 std::shared_ptr<horus_msgs::srv::LoadCalibration::Response> res)
490 {
491 if (!plugin_ || !store_)
492 {
493 res->success = false;
494 res->message = "node not configured or has no calibration plugin";
495 return;
496 }
497 if (req->path.empty())
498 {
499 const std::string subject =
500 req->subject_id.empty() ? get_parameter("subject_id").as_string() : req->subject_id;
501 res->units = loadLatest(subject, res->message);
502 res->success = !res->units.empty();
503 return;
504 }
505 std::lock_guard<std::mutex> lock(mutex_);
506 try
507 {
508 const auto doc = CalibrationStore::read(req->path);
509 const auto unit = doc["unit"].as<std::string>();
510 if (std::find(channels_.begin(), channels_.end(), unit) == channels_.end())
511 throw std::runtime_error("unit '" + unit + "' is not a channel of this node");
512 if (!plugin_->load(unit, doc["data"])) throw std::runtime_error("plugin rejected the data");
513 sources_[unit] = req->path;
514 res->units.push_back(unit);
515 res->success = true;
516 res->message = "loaded " + unit;
517 }
518 catch (const std::exception& e)
519 {
520 res->success = false;
521 res->message = e.what();
522 }
523 }

525 std::vector<std::string> loadLatest(const std::string& subject, std::string& message)
526 {
527 std::lock_guard<std::mutex> lock(mutex_);
528 std::vector<std::string> loaded;
529 if (subject.empty())
530 {
531 message = "no subject_id";
532 return loaded;
533 }
534 for (const auto& ch : channels_)
535 {
536 const auto path = store_->latest(subject, get_name(), ch);
537 if (!path) continue;
538 try
539 {
540 if (plugin_->load(ch, CalibrationStore::read(*path)["data"]))
541 {
542 sources_[ch] = *path;
543 loaded.push_back(ch);
544 }
545 }
546 catch (const std::exception& e)
547 {
548 RCLCPP_WARN(get_logger(), "skipping %s: %s", path->c_str(), e.what());
549 }
550 }
551 message = "loaded " + std::to_string(loaded.size()) + "/" + std::to_string(channels_.size()) + " channels";
552 return loaded;
553 }

555 /* -- Diagnostics ----------------------------------------------------------------------- */
556
557 void reportDiagnostics(diagnostic_updater::DiagnosticStatusWrapper& stat)
558 {
559 std::lock_guard<std::mutex> lock(mutex_);
560 if (!plugin_)
561 {
562 stat.summary(diagnostic_msgs::msg::DiagnosticStatus::OK, "no calibration configured");
563 return;
564 }
565 std::size_t done = 0;
566 for (const auto& ch : channels_)
567 {
568 const bool cal = plugin_->calibrated(ch);
569 done += cal ? 1 : 0;
570 stat.add(ch, cal ? sources_[ch] : std::string("not calibrated"));
571 }
572 stat.add("plugin", plugin_name_);
573 stat.add("mode", mode_ == CALIBRATED ? "calibrated" : mode_ == CONDITIONED ? "conditioned" : "bypass");
574 stat.add("calibrating", calibrating_ ? "yes" : "no");
575 if (done == channels_.size())
576 stat.summary(diagnostic_msgs::msg::DiagnosticStatus::OK, "all channels calibrated");
577 else
578 stat.summaryf(diagnostic_msgs::msg::DiagnosticStatus::WARN, "%zu/%zu channels calibrated", done,
579 channels_.size());
580 }
