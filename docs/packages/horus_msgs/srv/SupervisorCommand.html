SupervisorCommand

This is a ROS service definition.

Source

# Operator request to the supervisor. Requests are arbitrated: a request not allowed in the current
# mode is rejected (success = false) and the mode does not change.
uint8 START          = 0   # STANDBY -> RUN, running `tree` (the protocol) or trees.run
uint8 STOP           = 1   # RUN / MANUAL / INIT -> STANDBY (halts the tree; ends a per-run trial)
uint8 MANUAL         = 2   # STANDBY -> MANUAL (the operator's joint sliders move the device)
uint8 RESET          = 3   # EMERGENCY_STOP -> STANDBY (refused while a fault is active)
uint8 EMERGENCY_STOP = 4   # any -> EMERGENCY_STOP
uint8 command
string tree                # START: protocol tree to run instead of trees.run
---
bool success
string message
