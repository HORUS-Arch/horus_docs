58template <class SrvT>
59class ServiceLeaf : public BT::StatefulActionNode
60{
61 public:
62 ServiceLeaf(const std::string& name, const BT::NodeConfig& config, const NodePtr& node, const std::string& service)
63 : BT::StatefulActionNode(name, config), node_(node)
64 {
65 group_ = node_->create_callback_group(rclcpp::CallbackGroupType::MutuallyExclusive, false);
66 exec_.add_callback_group(group_, node_->get_node_base_interface());
67 client_ = node_->create_client<SrvT>(service, rclcpp::ServicesQoS(), group_);
68 }
69
70 static BT::PortsList basePorts() { return {BT::InputPort<double>("timeout", 2.0, "s to wait for the reply")}; }
71
72 BT::NodeStatus onStart() override
73 {
74 auto req = std::make_shared<typename SrvT::Request>();
75 if (!fill(*req)) return BT::NodeStatus::FAILURE;
76 if (!client_->service_is_ready())
77 {
78 RCLCPP_ERROR(node_->get_logger(), "%s: %s not available", registrationName().c_str(), client_->get_service_name());
79 return BT::NodeStatus::FAILURE;
80 }
81 deadline_ = std::chrono::steady_clock::now() +
82 std::chrono::duration_cast<std::chrono::steady_clock::duration>(
83 std::chrono::duration<double>(getInput<double>("timeout").value_or(2.0)));
84 future_ = client_->async_send_request(req).future.share();
85 return BT::NodeStatus::RUNNING;
86 }
87
88 BT::NodeStatus onRunning() override
89 {
90 exec_.spin_all(std::chrono::milliseconds(5));
91 if (future_.wait_for(std::chrono::seconds(0)) == std::future_status::ready)
92 {
93 const auto res = future_.get();
94 if (!res->success)
95 RCLCPP_WARN(node_->get_logger(), "%s: %s", registrationName().c_str(), res->message.c_str());
96 return res->success ? BT::NodeStatus::SUCCESS : BT::NodeStatus::FAILURE;
97 }
98 if (std::chrono::steady_clock::now() > deadline_)
99 {
100 RCLCPP_ERROR(node_->get_logger(), "%s: no reply", registrationName().c_str());
101 return BT::NodeStatus::FAILURE;
102 }
103 return BT::NodeStatus::RUNNING;
104 }
105
106 void onHalted() override { exec_.spin_all(std::chrono::milliseconds(1)); }
107
108 protected:
110 virtual bool fill(typename SrvT::Request& req) = 0;
111
112 NodePtr node_;
113
114 private:
115 rclcpp::CallbackGroup::SharedPtr group_;
116 rclcpp::executors::SingleThreadedExecutor exec_;
117 typename rclcpp::Client<SrvT>::SharedPtr client_;
118 std::shared_future<typename SrvT::Response::SharedPtr> future_;
119 std::chrono::steady_clock::time_point deadline_;
120};

176class SetSpeedScaleNode : public BT::StatefulActionNode
177{
178 public:
179 SetSpeedScaleNode(const std::string& name, const BT::NodeConfig& config, const NodePtr& node)
180 : BT::StatefulActionNode(name, config), node_(node)
181 {
182 group_ = node_->create_callback_group(rclcpp::CallbackGroupType::MutuallyExclusive, false);
183 exec_.add_callback_group(group_, node_->get_node_base_interface());
184 }
185
186 static BT::PortsList providedPorts()
187 {
188 return {BT::InputPort<double>("value", "speed scale (0 = hold, 1 = nominal)"),
189 BT::InputPort<std::string>("generator", "", "generator server node; empty = from `generator_action`")};
190 }
191
192 BT::NodeStatus onStart() override
193 {
194 const auto value = getInput<double>("value");
195 if (!value) return BT::NodeStatus::FAILURE;
196 auto target = getInput<std::string>("generator").value_or("");
197 if (target.empty())
198 {
199 target = node_->get_parameter("generator_action").as_string(); /* <server>/execute_behavior */
200 target = target.substr(0, target.rfind('/'));
201 }
202 if (!client_ || target != target_)
203 {
204 client_ = std::make_shared<rclcpp::AsyncParametersClient>(node_, target, rclcpp::ParametersQoS().get_rmw_qos_profile(), group_);
205 target_ = target;
206 }
207 if (!client_->service_is_ready())
208 {
209 RCLCPP_ERROR(node_->get_logger(), "SetSpeedScale: %s parameters not reachable", target.c_str());
210 return BT::NodeStatus::FAILURE;
211 }
212 future_ = client_->set_parameters({rclcpp::Parameter("speed_scale", *value)});
213 return BT::NodeStatus::RUNNING;
214 }
215
216 BT::NodeStatus onRunning() override
217 {
218 exec_.spin_all(std::chrono::milliseconds(5));
219 if (future_.wait_for(std::chrono::seconds(0)) != std::future_status::ready) return BT::NodeStatus::RUNNING;
220 const auto results = future_.get();
221 return !results.empty() && results.front().successful ? BT::NodeStatus::SUCCESS : BT::NodeStatus::FAILURE;
222 }
223
224 void onHalted() override {}
225
226 private:
227 NodePtr node_;
228 rclcpp::CallbackGroup::SharedPtr group_;
229 rclcpp::executors::SingleThreadedExecutor exec_;
230 std::shared_ptr<rclcpp::AsyncParametersClient> client_;
231 std::string target_;
232 std::shared_future<std::vector<rcl_interfaces::msg::SetParametersResult>> future_;
233};
